C17 Embedding SDK (`bpp_api.h`)

#include "bpp_api.h"

int main(void) {
    BppEngineContext *ctx = bpp_init(671088640L);
    bpp_exec_string(ctx, "PRINT \"Hello from Embedded BASIC++!\"");
    bpp_shutdown(ctx);
    return 0;
}
